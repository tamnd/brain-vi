## Giải pháp Python```python
import sys
input = sys.stdin.readline

def gauss_xor(mat, n, m):
    row = 0
    for col in range(m):
        pivot = -1
        for i in range(row, n):
            if mat[i][col]:
                pivot = i
                break
        if pivot == -1:
            continue
        mat[row], mat[pivot] = mat[pivot], mat[row]

        for i in range(n):
            if i != row and mat[i][col]:
                for j in range(col, m + 1):
                    mat[i][j] ^= mat[row][j]

        row += 1
        if row == n:
            break

def main():
    n = int(input().strip())
    mat = []

    for _ in range(n):
        row = list(map(int, input().split()))
        mat.append(row)

    m = n

    gauss_xor(mat, n, m)

    ans = [0] * n
    for i in range(n):
        for j in range(n):
            if mat[i][j]:
                ans[j] = mat[i][n]
                break

    print(*ans)

if __name__ == "__main__":
    main()
```Việc triển khai xây dựng một biểu diễn ma trận nhị phân của hệ thống truy vấn và giảm bớt nó bằng cách sử dụng phép loại bỏ Gaussian dựa trên XOR.
